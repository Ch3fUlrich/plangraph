#!/usr/bin/env python3
"""Coding-side fleet-bus poller (pull mode, D-2024/D-2029; peer discovery D-2670).

Outbound-only: coding.vm polls the workstation listener, so no inbound
firewall hole and no coding listener are needed. Mirrors the workstation
`tools/fleet_bus.py` contract exactly:

  POST /inbox  JSON {"from","to","ts","nonce","text"} + header X-Fleet-Sig
               = hex HMAC-SHA256(secret, "ts|nonce|from|to|sha256(text)")
  GET /outbox?since=<cursor> -> 200 {"cursor", "records"} (records carry text)

The peer endpoint is DISCOVERED, never guessed (BUS-PEER-DISCOVERY-ADOPT):

  GET <keeper-url>/v1/hosts   header "Authorization: Bearer <token>"
      -> 200 {"hosts":[{"host","endpoint","key_id","health","last_seen"}]}

  The target host's endpoint is used only when health == "healthy" and
  endpoint is non-null. Anything else (keeper URL/token unset, keeper
  unreachable, non-200, bad JSON, host unknown, host unhealthy, endpoint
  null) fails closed: one line on stderr, exit 3, no traceback, and never a
  fallback IP or loopback guess. `--base URL` (env FLEET_BUS_BASE) is the
  explicit transition override and skips discovery; it stays opt-in.

Configuration (flag / env; no literal defaults anywhere):

  --host NAME / FLEET_BUS_HOST                  peer to resolve (required
                                                unless --base is given)
  --base URL / FLEET_BUS_BASE                   explicit peer, skips discovery
  --keeper-url URL / FLEET_KEEPER_URL           keeper base URL
  --keeper-token-file PATH / FLEET_KEEPER_TOKEN_FILE
                                                0600 bearer token file
  --inventory PATH / FLEET_BUS_INVENTORY        per-host key inventory JSON
                                                {"hosts":{"<host>":{"key_file":"<path>"}}}
  --secret-file PATH                            explicit override only: the
                                                legacy shared fleet-relay-secret

Secrets travel by NAME and FILE only, never in output: `send`/`poll` sign with
the per-host key file named by the inventory for --host (read once, first line,
owner-only 0600); the old shared secret file is reachable only through an
explicit --secret-file. `probe` needs no key (open read, reachability only).

Exit codes: 0 ok; 2 usage/config error (host, inventory, key file);
3 fail-closed (keeper/discovery/transport); 4 auth refused.
"""

import argparse
import hashlib
import hmac
import json
import os
import sys
import time
import urllib.error
import urllib.request
import uuid

TIMEOUT_S = 10
TEXT_MAX = 2000
CURSOR_FILE = os.path.expanduser("~/.local/state/fleet/fleet-poll.cursor")


def die(msg, code=2):
    """One line on stderr, no traceback, then exit with `code`."""
    sys.stderr.write("fleet-poll: error: %s\n" % msg)
    raise SystemExit(code)


def scrub_text(text):
    """One line, at most TEXT_MAX chars. Mirrors the listener."""
    return " ".join(str(text).splitlines())[:TEXT_MAX]


def read_secret(path, what="secret file", code=2):
    """First line of the secret file, owner-only (0600). Names the failure,
    never the value."""
    try:
        mode = os.stat(path).st_mode
        with open(path, encoding="utf-8-sig") as f:
            value = f.readline().strip()
    except OSError:
        die("%s unreadable: %s" % (what, path), code)
    if mode & 0o077:
        die("%s not owner-only (0600): %s" % (what, path), code)
    if not value:
        die("%s empty: %s" % (what, path), code)
    return value.encode("utf-8")


def sign(secret, ts, nonce, frm, to, text):
    """Hex HMAC-SHA256 over ts|nonce|from|to|sha256(text). Secret stays bytes."""
    digest = hashlib.sha256(str(text).encode("utf-8")).hexdigest()
    canonical = "%s|%s|%s|%s|%s" % (ts, nonce, frm, to, digest)
    return hmac.new(secret, canonical.encode("utf-8"),
                    hashlib.sha256).hexdigest()


def sig_headers(secret):
    """Signed request headers for the open GET: same canonical form with the
    empty message (from, to and text unset)."""
    ts = int(time.time())
    nonce = uuid.uuid4().hex
    return {"X-Fleet-Ts": str(ts), "X-Fleet-Nonce": nonce,
            "X-Fleet-Sig": sign(secret, ts, nonce, "", "", "")}


def _request(method, url, body=None, headers=None):
    req = urllib.request.Request(url, data=body, headers=headers or {},
                                 method=method)
    try:
        with urllib.request.urlopen(req, timeout=TIMEOUT_S) as resp:
            return resp.status, resp.read().decode("utf-8")
    except urllib.error.HTTPError as e:
        if e.code in (401, 403):
            return e.code, ""
        die("HTTP %s on %s" % (e.code, url), 3)
    except OSError as e:
        die("transport: %s" % e, 3)


def discover_endpoint(host, keeper_url, token_file):
    """The host's endpoint from the keeper registry. Fails closed (exit 3)
    unless the host is healthy and its endpoint is non-null."""
    if not keeper_url:
        die("keeper URL unset (--keeper-url / FLEET_KEEPER_URL)", 3)
    if not token_file:
        die("keeper token file unset "
            "(--keeper-token-file / FLEET_KEEPER_TOKEN_FILE)", 3)
    token = read_secret(token_file, "keeper token file", 3)
    url = keeper_url.rstrip("/") + "/v1/hosts"
    req = urllib.request.Request(
        url, headers={"Authorization": "Bearer " + token.decode("utf-8")})
    try:
        with urllib.request.urlopen(req, timeout=TIMEOUT_S) as resp:
            payload = resp.read().decode("utf-8", "replace")
    except urllib.error.HTTPError as e:
        die("keeper HTTP %s on %s" % (e.code, url), 3)
    except OSError:
        die("keeper unreachable: %s" % keeper_url, 3)
    try:
        hosts = json.loads(payload)["hosts"]
    except (ValueError, KeyError, TypeError):
        die("bad keeper payload from %s" % url, 3)
    if not isinstance(hosts, list):
        die("bad keeper payload from %s" % url, 3)
    for entry in hosts:
        if not isinstance(entry, dict) or entry.get("host") != host:
            continue
        if entry.get("health") != "healthy":
            die("host not healthy in keeper: %s (health=%s)"
                % (host, entry.get("health")), 3)
        endpoint = entry.get("endpoint")
        if not endpoint:
            die("no endpoint for host in keeper: %s" % host, 3)
        return str(endpoint)
    die("unknown host in keeper: %s" % host, 3)


def resolve_base(args):
    """--base (or FLEET_BUS_BASE) wins and skips discovery; otherwise the
    peer comes from the keeper registry for --host (or FLEET_BUS_HOST)."""
    base = getattr(args, "base", None) or os.environ.get("FLEET_BUS_BASE") or ""
    if base:
        return base.rstrip("/")
    if not args.host:
        die("--host (or FLEET_BUS_HOST) required unless --base "
            "(or FLEET_BUS_BASE) is given", 2)
    keeper_url = (getattr(args, "keeper_url", None)
                  or os.environ.get("FLEET_KEEPER_URL") or "")
    token_file = (getattr(args, "keeper_token_file", None)
                  or os.environ.get("FLEET_KEEPER_TOKEN_FILE") or "")
    return discover_endpoint(args.host, keeper_url, token_file).rstrip("/")


def load_key(args):
    """The signing key: explicit --secret-file override, else the per-host key
    file named by the inventory for --host. Read once; every failure names the
    path and host, never the contents."""
    secret_file = getattr(args, "secret_file", None)
    if secret_file:
        return read_secret(secret_file, "secret file", 2)
    if not args.host:
        die("--host (or FLEET_BUS_HOST) required for key lookup", 2)
    inv_path = (getattr(args, "inventory", None)
                or os.environ.get("FLEET_BUS_INVENTORY") or "")
    if not inv_path:
        die("inventory unset (--inventory / FLEET_BUS_INVENTORY)", 2)
    try:
        with open(inv_path, encoding="utf-8") as f:
            inv = json.load(f)
    except (OSError, ValueError):
        die("inventory unreadable or bad JSON: %s" % inv_path, 2)
    hosts = inv.get("hosts") if isinstance(inv, dict) else None
    entry = hosts.get(args.host) if isinstance(hosts, dict) else None
    key_file = entry.get("key_file") if isinstance(entry, dict) else None
    if not key_file:
        die("no key_file for host %s in inventory %s" % (args.host, inv_path), 2)
    return read_secret(key_file, "key file for host %s" % args.host, 2)


def cmd_probe(args):
    """Reachability only: GET /outbox?since= (no secret). Prints one line."""
    status, payload = _request("GET", args.base + "/outbox?since=")
    if status == 401:
        print("fleet-poll: probe refused (auth)")
        return 4
    try:
        data = json.loads(payload)
        print("fleet-poll: probe ok: cursor=%s records=%d"
              % (data.get("cursor", "?"), len(data.get("records", []))))
    except (ValueError, AttributeError):
        die("bad probe payload", 2)
    return 0


def cmd_poll(args):
    """Fetch records since the stored cursor, print them, advance the cursor.
    Signed with the per-host key file (probe stays unsigned)."""
    secret = load_key(args)
    # The cursor file holds one cursor line; absent means fetch-all.
    try:
        with open(args.cursor_file, encoding="utf-8") as f:
            cursor = f.readline().strip()
    except OSError:
        cursor = ""
    status, payload = _request(
        "GET", args.base + "/outbox?since=" + urllib.request.quote(cursor),
        headers=sig_headers(secret))
    if status == 401:
        print("fleet-poll: poll refused (auth)")
        return 4
    try:
        data = json.loads(payload)
    except ValueError:
        die("bad poll payload", 2)
    for rec in data.get("records", []):
        print(json.dumps({"id": rec.get("id"), "ts": rec.get("ts"),
                          "from": rec.get("from"), "to": rec.get("to"),
                          "text": scrub_text(rec.get("text", ""))},
                         ensure_ascii=False))
    try:
        os.makedirs(os.path.dirname(args.cursor_file), exist_ok=True)
        with open(args.cursor_file, "w", encoding="utf-8") as f:
            f.write(str(data.get("cursor", cursor)) + "\n")
    except OSError:
        die("cursor file unwritable: %s" % args.cursor_file, 2)
    return 0


def cmd_send(args):
    """POST one message to /inbox, signed with the per-host key file."""
    secret = load_key(args)
    ts = int(time.time())
    nonce = uuid.uuid4().hex
    text = scrub_text(args.text)
    body = json.dumps({"from": args.frm, "to": args.to, "ts": ts,
                       "nonce": nonce, "text": text}).encode("utf-8")
    sig = sign(secret, ts, nonce, args.frm, args.to, text)
    status, _ = _request("POST", args.base + "/inbox", body=body,
                         headers={"Content-Type": "application/json",
                                  "X-Fleet-Sig": sig})
    if status == 401:
        print("fleet-poll: send refused (bad signature, skew or replay)")
        return 4
    print("fleet-poll: sent ok: ts=%d nonce=%.8s..." % (ts, nonce))
    return 0


def _add_common(ap, hidden=False):
    """Peer/discovery options on the main parser and on every subparser (so
    `fleet-poll.py --host ws probe` and `fleet-poll.py probe --host ws` both
    work); subparser copies are SUPPRESS so they never clobber the outer value."""
    default = argparse.SUPPRESS if hidden else None
    ap.add_argument("--base", default=default,
                    help="explicit peer base URL; skips discovery "
                         "(env FLEET_BUS_BASE)")
    ap.add_argument("--host", default=default,
                    help="peer host name to resolve in the keeper registry "
                         "(env FLEET_BUS_HOST)")
    ap.add_argument("--keeper-url", default=default,
                    help="keeper base URL for GET /v1/hosts (env FLEET_KEEPER_URL)")
    ap.add_argument("--keeper-token-file", default=default,
                    help="0600 file holding the keeper bearer token "
                         "(env FLEET_KEEPER_TOKEN_FILE)")
    ap.add_argument("--inventory", default=default,
                    help="per-host key inventory JSON (env FLEET_BUS_INVENTORY)")


def build_parser():
    ap = argparse.ArgumentParser(prog="fleet-poll.py",
                                 description="Coding-side fleet-bus poller.")
    _add_common(ap)
    sub = ap.add_subparsers(dest="cmd", required=True)
    p = sub.add_parser("probe", help="reachability check, no secret")
    _add_common(p, hidden=True)
    p.set_defaults(func=cmd_probe)
    p = sub.add_parser("poll", help="fetch new records, advance cursor")
    _add_common(p, hidden=True)
    p.add_argument("--cursor-file", default=CURSOR_FILE)
    p.add_argument("--secret-file",
                   help="explicit override: legacy shared secret file")
    p.set_defaults(func=cmd_poll)
    p = sub.add_parser("send", help="POST one signed message")
    _add_common(p, hidden=True)
    p.add_argument("--secret-file",
                   help="explicit override: legacy shared secret file")
    p.add_argument("--from", dest="frm", required=True)
    p.add_argument("--to", required=True)
    p.add_argument("--text", required=True)
    p.set_defaults(func=cmd_send)
    return ap


def main(argv=None):
    args = build_parser().parse_args(argv)
    args.host = getattr(args, "host", None) or os.environ.get("FLEET_BUS_HOST") or ""
    args.inventory = (getattr(args, "inventory", None)
                      or os.environ.get("FLEET_BUS_INVENTORY") or "")
    args.base = resolve_base(args)
    return args.func(args)


if __name__ == "__main__":
    sys.exit(main())
